% Level scheme of the two-photon Raman transition in Rb-85 with all quantities of
%   Delta_e = Delta + (nu_D1 - nu_J') - Delta_hfs(J',F')
% (all as frequencies nu = omega/2pi; NOT to scale)
% Requires: tikz (library arrows.meta), siunitx, amsmath
\begin{tikzpicture}[
    >={Stealth[length=1.8mm]},
    lvl/.style={line width=1.1pt},
    sub/.style={line width=0.8pt},
    subthin/.style={line width=0.5pt, gray},
    virt/.style={line width=0.9pt, dashed},
    guide/.style={line width=0.4pt, densely dotted},
    leg/.style={->, line width=1.1pt},
    det/.style={<->, line width=0.6pt},
    brk/.style={line width=0.6pt},
    x=1cm, y=1cm]

  % ================= levels =================
  % 5S_1/2
  \draw[lvl] (1.2,-1.2) -- (10.6,-1.2) node[right] {$F=2$};
  \draw[lvl] (1.2,-0.2) -- (10.6,-0.2) node[right] {$F=3$};
  \node[left] at (1.0,-0.7) {$5S_{1/2}$};

  % 5P_1/2 (D1)
  \draw[guide] (1.2,4.0) -- (10.6,4.0) node[right] {$\nu_{1/2}=\nu_{\mathrm{D}_1}$};
  \draw[sub] (4.6,3.4) -- (10.6,3.4) node[right] {$F'=2$};
  \draw[sub] (4.6,4.6) -- (10.6,4.6) node[right] {$F'=3$};
  \node[left, align=right] at (1.0,4.0) {$5P_{1/2}$\\ (D$_1$)};

  % virtual level
  \draw[virt] (1.2,6.2) -- (10.6,6.2) node[right] {$\nu_{\mathrm{D}_1}+\Delta$};

  % 5P_3/2 (D2)
  \draw[subthin] (4.6,8.70) -- (10.6,8.70);
  \draw[subthin] (4.6,8.95) -- (10.6,8.95);
  \node[right, gray] at (10.6,8.82) {$F'=1,2$};
  \draw[sub] (4.6,9.30) -- (10.6,9.30) node[right] {$F'=3$};
  \draw[guide] (1.2,9.8) -- (10.6,9.8) node[right] {$\nu_{3/2}=\nu_{\mathrm{D}_2}$};
  \draw[subthin] (4.6,10.5) -- (10.6,10.5) node[right] {$F'=4$};
  \node[left, align=right] at (1.0,9.8) {$5P_{3/2}$\\ (D$_2$)};

  % ================= Raman legs =================
  \coordinate (V) at (3.6,6.2);
  \draw[leg, blue!70!black]   (2.5,-1.2) -- (V) node[pos=0.45, left] {$\omega_1$};
  \draw[leg, orange!85!black] (4.5,-0.2) -- (V) node[pos=0.45, right] {$\omega_2$};
  \fill (V) circle (1.6pt);

  % ================= quantities =================
  % Delta: virtual level above the D1 centroid
  \draw[det] (1.5,4.0) -- (1.5,6.2)
        node[midway, right] {$\Delta$};

  % --- D1, F'=3 ---
  \draw[det] (4.9,4.0) -- (4.9,4.6)
        node[midway, right, font=\small] {$\Delta_\text{hfs}(F'{=}3)$};
  \draw[det, red!75!black, line width=0.9pt] (5.6,4.6) -- (5.6,6.2)
        node[midway, right, red!75!black] {$\Delta_{e}^{\mathrm{D}_1,F'=3}$};

  % --- D2 line offset: nu_D2 - nu_D1 = -(nu_D1 - nu_J') ---
  \draw[det] (7.9,4.0) -- (7.9,9.8)
        node[pos=0.5, right] {$\nu_{\mathrm{D}_2}-\nu_{\mathrm{D}_1}$};
  % --- D2, F'=3 ---
  \draw[det] (4.9,9.3) -- (4.9,9.8)
        node[midway, right, font=\small] {$\Delta_\text{hfs}(F'{=}3)$};
  \draw[det, red!75!black, line width=0.9pt] (10.1,6.2) -- (10.1,9.3)
        node[pos=0.63, left, red!75!black] {$\Delta_{e}^{\mathrm{D}_2,F'=3}$};

  % axis breaks on the long arrows
  \foreach \xx in {7.9, 10.1}{
    \fill[white] (\xx-0.22,7.70) rectangle (\xx+0.22,7.95);
    \draw[brk] (\xx-0.22,7.66) -- (\xx+0.22,7.80);
    \draw[brk] (\xx-0.22,7.86) -- (\xx+0.22,8.00);
  }

  % ================= formula and numbers =================
  \node[anchor=north west, align=left] at (5.0,3.05) {%
    $\Delta_e=\Delta+(\nu_{\mathrm{D}_1}-\nu_{J'})-\Delta_\text{hfs}(J',F')$\\[4pt]
    D$_1$ ($J'{=}1/2$): $\nu_{\mathrm{D}_1}-\nu_{J'}=0$\\[2pt]
    D$_2$ ($J'{=}3/2$): $\nu_{\mathrm{D}_1}-\nu_{J'}=-\SI{7122.92}{\giga\hertz}$\\[4pt]
    D$_1$, $F'{=}3$: $\;50-0-0.151=\SI{49.85}{\giga\hertz}$\\[2pt]
    D$_2$, $F'{=}3$: $\;50-7122.92+0.020=-\SI{7072.90}{\giga\hertz}$};
\end{tikzpicture}
